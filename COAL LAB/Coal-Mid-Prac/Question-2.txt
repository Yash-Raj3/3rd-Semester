INCLUDE Irvine32.inc

.data
arr DWORD 101, 202, 303, 404, 505, 606, 707, 808, 909, 111, 222, 333,221,123,124,125
reversed  DWORD 16  DUP(?)
mesg Byte "The Original array: ",0
mesg2 byte "The Reversed array: ",0

.code
main PROC
mov edx,offset mesg
call WriteString 

mov ecx,lengthof arr
mov esi,offset arr
displayOriginal:
mov eax,[esi]
call writedec
mov al,' '
call WriteChar
add esi,4
loop displayOriginal
call crlf
; Reversing 
mov esi,0
mov edi,lengthof arr -1
mov ecx,lengthof arr




reverse:
mov eax ,arr[edi* type arr]
mov reversed[esi* type reversed],eax
inc esi
dec edi
loop reverse


mov edx,offset mesg2
call WriteString 
mov ecx,lengthof reversed
mov esi,offset reversed
displayReversed:
mov eax,[esi]
call writedec
mov al,' '
call WriteChar
add esi,4
loop displayReversed

  
  

    exit
main ENDP
END main
