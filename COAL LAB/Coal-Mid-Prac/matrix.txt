INCLUDE Irvine32.inc

.data
rows=  3 
cols= 4
matrix Dword 2,4,6,8
       Dword 1,3,5,7
       Dword 10,20,30,40
col byte "Sum of Col = ",0

.code 
main proc


mov ecx,rows
mov esi,0
outer:
push ecx
mov ecx,cols
inner:
mov eax,matrix[esi*type matrix]
call writedec
mov al,' '
call writechar
inc esi
loop inner
call crlf
pop ecx
loop outer

call crlf
mov edi,0
mov edi,0
mov ecx, cols

mov ebx,0
o:
push ecx 
mov ecx,rows
mov eax, 0
mov esi,edi
i:
add eax,matrix[esi *type matrix]
add esi,cols
loop i
mov edx,offset col[0]
call writestring
call writedec
call crlf
add ebx,16
inc edi
inc ebx
pop ecx
loop o









    exit
main ENDP
END main
